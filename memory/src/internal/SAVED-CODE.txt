#ifndef MEMORY_MEM_STRUCTS_H
#define MEMORY_MEM_STRUCTS_H

enum MEMORY_ERROR_CODE {
    NO_MEMORY_MANAGER_INIT = BIT_MASK_FROM_UINT(1),
    NO_FREE_MEMORY = BIT_MASK_FROM_UINT(2),
    MEMMAN_ALREADY_INITIALIZED = BIT_MASK_FROM_UINT(3),
    ALLOC_NOT_CONTIGUOUS_MEM = BIT_MASK_FROM_UINT(4),
    MEMORY_IS_IN_USE = BIT_MASK_FROM_UINT(5),
};

#define NUM_CHUNKS 16
#define PAGES_PER_CHUNK 32

#if  PAGES_PER_CHUNK == 32
#define CHUNK_ID_BIT_SIZE 6
#else
#define CHUNK_ID_BIT_SIZE 8
#endif

#define GET_ADDR_OF_MEMTABLE(memtable) (memtable)
#define GET_ADDR_OF_CHUNK_BUF(memtable, CHUNK_ID) (&memtable->chunks_buf+CHUNK_ID)
#define CHUNK_GET_ADDR_FROM_CHUNK_BUF_BY_ID(memtable, CHUNK_ID)\
                ( (char*) (memtable->chunks_buf) + (sizeof(struct mem_chunk*) * NUM_CHUNKS) )

struct mem_chunk; // all allocated mem chunks

struct mem_table {
    void* mem_start;                // allocated start address
    struct mem_chunk* chunks_buf;   // all allocated mem chunks
    uint16 capacity;                // number of chunks
    uint16 count;                   // chunks in use
};
struct mem_flags {
    uint32 is_init;
};
struct mem_used;
struct mem_chunk {
    uint16 id;
    uint16 num_pages;
    struct mem_flags flags;
    struct mem_used* mem_used_buf;
};
struct mem_used {
    uint32 chunk_id: CHUNK_ID_BIT_SIZE; // which chunk belongs to
    uint32 page_id: CHUNK_ID_BIT_SIZE;  // the id of the first page of the registered used memory
};

static struct mem_chunk* chunks_buffer;
static struct mem_used* mem_used_buffer;


static struct mem_table* mem_table;

static void mem_init()
{
    const uint32 pagesize = 8;
    const uint32 num_pages = 512;
    const uint32 alignment = PAGE_SIZE * 16;
    const uint32 self_store_size = 10*PAGE_SIZE;

    void* addr = aligned_alloc(alignment, self_store_size + (num_pages * pagesize));
    printf("addr:%p\n", addr);

    mem_table = addr;
    mem_table->mem_start = addr;

    addr += sizeof(struct mem_table);

    mem_table->chunks_buf = addr;

    mem_table->capacity = 0xabcd;
    mem_table->count = 0xdcba;
    for(uint32 i = 0; i < NUM_CHUNKS; i++) {
        mem_table->chunks_buf[i].id = i;
        mem_table->chunks_buf[i].num_pages = 0xdddd;
        mem_table->chunks_buf[i].flags.is_init = 0xeeee;
    }

    printf("mem addr:%p\n", &mem_table);
    printf("chunk addr:%p\n", &(mem_table->chunks_buf));

    printf("GET_ADDR_OF_MEMTABLE:%p\n", GET_ADDR_OF_MEMTABLE(mem_table));
    printf("GET_ADDR_OF_CHUNK_BUF:%p\n", GET_ADDR_OF_CHUNK_BUF(mem_table, 0));

    /*************************************************************
     * Set storage layout for all chunks struct in chunks_buf
     *************************************************************/
    void* p = CHUNK_GET_ADDR_FROM_CHUNK_BUF_BY_ID(mem_table, 0);
    for(uint32 id = 0; id < NUM_CHUNKS; id++) {

        char* chk = (char*) p + (PAGES_PER_CHUNK*id);
        for(uint32 i = 0; i < 8; i++) { // add 8 chars at the start of each chunk
            chk[i] = i + 'A';
            printf("page [%u]: %c\n", i, chk[i]);
        }
    }


    /*************************************************************
     * Set storage layout for used memory struct
     *************************************************************/

}
